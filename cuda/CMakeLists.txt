cmake_minimum_required(VERSION 3.22)
project(mgbfs_cuda LANGUAGES CXX CUDA)
set(CUTLASS_ROOT /opt/cutlass CACHE PATH "Pinned CUTLASS source checkout")
option(MGBFS_NCCL_LSA "Build device-count NCCL LSA transport" OFF)
option(MGBFS_NVTX "Enable explicit runtime profiling ranges" OFF)
set(MGBFS_NVTX_INCLUDE_DIR "" CACHE PATH "NVTX3 include directory")
set(MGBFS_NCCL_ROOT "" CACHE PATH "NCCL 2.29+ package root for LSA transport")
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 75)
endif()
add_library(mgbfs_cuda SHARED allocation_shape.cpp hash.cu generate.cu route.cu owner.cu macro_settle.cu future_merge.cu exchange_pack.cu materialize.cu bounded_owner.cu bounded_owner_query.cpp state_commit.cu archive_pack.cu directories.cu nccl_transport.cpp)
set_source_files_properties(nccl_transport.cpp PROPERTIES LANGUAGE CUDA)
target_sources(mgbfs_cuda PRIVATE regenerate.cu hash_first_generate.cu generic_graph.cu generic_visited.cu generic_route.cu generic_sorted_native.cu)
target_sources(mgbfs_cuda PRIVATE graph_window.cu shard_ab_job.cu shard_ab_pipeline.cu)
target_sources(mgbfs_cuda PRIVATE profile_ranges.cpp)
if(MGBFS_NVTX)
  if(NOT EXISTS "${MGBFS_NVTX_INCLUDE_DIR}/nvtx3/nvToolsExt.h")
    message(FATAL_ERROR "MGBFS_NVTX requires an explicit NVTX3 include directory")
  endif()
  target_include_directories(mgbfs_cuda PRIVATE "${MGBFS_NVTX_INCLUDE_DIR}")
  target_compile_definitions(mgbfs_cuda PRIVATE MGBFS_NVTX=1)
  target_link_libraries(mgbfs_cuda PRIVATE ${CMAKE_DL_LIBS})
endif()
target_compile_features(mgbfs_cuda PRIVATE cxx_std_17 cuda_std_17)
target_include_directories(mgbfs_cuda PRIVATE ${CUTLASS_ROOT}/include)
if(MGBFS_NCCL_LSA)
  if(NOT EXISTS "${MGBFS_NCCL_ROOT}/include/nccl_device.h" OR
     NOT EXISTS "${MGBFS_NCCL_ROOT}/lib/libnccl.so.2")
    message(FATAL_ERROR "MGBFS_NCCL_LSA requires pinned NCCL include/nccl_device.h and lib/libnccl.so.2")
  endif()
  target_include_directories(mgbfs_cuda PRIVATE "${MGBFS_NCCL_ROOT}/include")
  target_link_libraries(mgbfs_cuda PRIVATE "${MGBFS_NCCL_ROOT}/lib/libnccl.so.2")
  target_compile_definitions(mgbfs_cuda PRIVATE MGBFS_NCCL_LSA=1)
  set_target_properties(mgbfs_cuda PROPERTIES BUILD_RPATH "${MGBFS_NCCL_ROOT}/lib")
else()
  if(MGBFS_NCCL_ROOT)
    if(NOT EXISTS "${MGBFS_NCCL_ROOT}/include/nccl.h" OR
       NOT EXISTS "${MGBFS_NCCL_ROOT}/lib/libnccl.so.2")
      message(FATAL_ERROR "MGBFS_NCCL_ROOT requires include/nccl.h and lib/libnccl.so.2")
    endif()
    target_include_directories(mgbfs_cuda PRIVATE "${MGBFS_NCCL_ROOT}/include")
    target_link_libraries(mgbfs_cuda PRIVATE "${MGBFS_NCCL_ROOT}/lib/libnccl.so.2")
    set_target_properties(mgbfs_cuda PROPERTIES BUILD_RPATH "${MGBFS_NCCL_ROOT}/lib")
  else()
    target_link_libraries(mgbfs_cuda PRIVATE nccl)
  endif()
endif()
target_compile_options(mgbfs_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo;--expt-relaxed-constexpr>)
set_target_properties(mgbfs_cuda PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
