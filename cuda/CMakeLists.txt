cmake_minimum_required(VERSION 3.22)
project(mgbfs_cuda LANGUAGES CXX CUDA)
set(CUTLASS_ROOT /opt/cutlass CACHE PATH "Pinned CUTLASS source checkout")
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 75)
endif()
add_library(mgbfs_cuda SHARED hash.cu generate.cu route.cu owner.cu materialize.cu)
target_compile_features(mgbfs_cuda PRIVATE cxx_std_17 cuda_std_17)
target_include_directories(mgbfs_cuda PRIVATE ${CUTLASS_ROOT}/include)
target_compile_options(mgbfs_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo;--expt-relaxed-constexpr>)
set_target_properties(mgbfs_cuda PROPERTIES POSITION_INDEPENDENT_CODE ON)
