cmake_minimum_required(VERSION 3.22)
project(mgbfs_cuda LANGUAGES CXX CUDA)
set(CUTLASS_ROOT /opt/cutlass CACHE PATH "Pinned CUTLASS source checkout")
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 75)
endif()
add_library(mgbfs_cuda SHARED allocation_shape.cpp hash.cu generate.cu route.cu owner.cu materialize.cu)
target_compile_features(mgbfs_cuda PRIVATE cxx_std_17 cuda_std_17)
target_include_directories(mgbfs_cuda PRIVATE ${CUTLASS_ROOT}/include)
target_compile_options(mgbfs_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo;--expt-relaxed-constexpr>)
set_target_properties(mgbfs_cuda PROPERTIES POSITION_INDEPENDENT_CODE ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(mgbfs-allocation-query-test ../tests/allocation_shape.cpp)
  target_include_directories(mgbfs-allocation-query-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(mgbfs-allocation-query-test PRIVATE PUBLIC_QUERY)
  target_compile_features(mgbfs-allocation-query-test PRIVATE cxx_std_17)
  target_link_libraries(mgbfs-allocation-query-test PRIVATE mgbfs_cuda)
  add_test(NAME allocation-query COMMAND mgbfs-allocation-query-test)
  add_executable(mgbfs-route-query-test ../tests/route_query.cpp)
  target_include_directories(mgbfs-route-query-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_features(mgbfs-route-query-test PRIVATE cxx_std_17)
  target_link_libraries(mgbfs-route-query-test PRIVATE mgbfs_cuda)
  add_test(NAME route-query COMMAND mgbfs-route-query-test)
endif()
