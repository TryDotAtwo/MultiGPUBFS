cmake_minimum_required(VERSION 3.30.4)
project(mgbfs_library_owner_probe LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
find_package(CUDAToolkit REQUIRED)
find_package(cudf 26.04.0 EXACT CONFIG REQUIRED)
find_package(rmm 26.04.0 EXACT CONFIG REQUIRED)
add_library(mgbfs_library_owner SHARED owner_abi.cpp layout.cu)
target_link_libraries(mgbfs_library_owner PUBLIC cudf::cudf rmm::rmm CUDA::cudart)
add_executable(cudf_owner_probe cudf_owner_probe.cpp ../../cuda/state_commit.cu)
target_link_libraries(cudf_owner_probe PRIVATE mgbfs_library_owner)
enable_testing()
add_executable(cuco_workspace_lease_contract ../../tests/cuco_workspace_lease_contract.cpp)
add_test(NAME cuco_workspace_lease_contract COMMAND cuco_workspace_lease_contract)
add_test(NAME cuco_workspace_lease_rejection_probe COMMAND cuco_workspace_lease_contract --probe-rejection)
set_tests_properties(cuco_workspace_lease_rejection_probe PROPERTIES WILL_FAIL TRUE)
add_test(NAME cudf_owner_probe COMMAND cudf_owner_probe)
add_executable(owner_abi_invalid ../../tests/library_owner_abi_invalid.cpp)
target_include_directories(owner_abi_invalid PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(owner_abi_invalid PRIVATE mgbfs_library_owner)
add_test(NAME owner_abi_invalid COMMAND owner_abi_invalid)
add_executable(owner_source_gather ../../tests/owner_source_gather.cu)
target_include_directories(owner_source_gather PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(owner_source_gather PRIVATE CUDA::cudart)
add_test(NAME owner_source_gather COMMAND owner_source_gather)

# Optional pinned cuco source supplied by the isolated GPU gate.
if(DEFINED CUCO_ROOT)

  target_include_directories(mgbfs_library_owner PRIVATE "${CUCO_ROOT}/include")
  add_executable(cuco_owner_abi_contract ../../tests/cuco_owner_abi_contract.cpp)
  target_link_libraries(cuco_owner_abi_contract PRIVATE mgbfs_library_owner)
  add_test(NAME cuco_owner_abi_contract COMMAND cuco_owner_abi_contract)
  add_executable(cuco_pool_probe cuco_pool_probe.cu)
  target_compile_options(cuco_pool_probe PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-extended-lambda>)
  target_include_directories(cuco_pool_probe PRIVATE "${CUCO_ROOT}/include")
  target_link_libraries(cuco_pool_probe PRIVATE rmm::rmm CUDA::cudart)
  add_test(NAME cuco_pool_probe COMMAND cuco_pool_probe)
  add_executable(cuco_owner_probe cuco_owner_probe.cu)
  target_compile_options(cuco_owner_probe PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-extended-lambda>)
  target_include_directories(cuco_owner_probe PRIVATE "${CUCO_ROOT}/include")
