cmake_minimum_required(VERSION 3.30.4)
project(mgbfs_library_owner_probe LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
find_package(CUDAToolkit REQUIRED)
find_package(cudf 26.04 EXACT CONFIG REQUIRED)
find_package(rmm 26.04 EXACT CONFIG REQUIRED)
add_executable(cudf_owner_probe cudf_owner_probe.cpp)
target_link_libraries(cudf_owner_probe PRIVATE cudf::cudf rmm::rmm CUDA::cudart)
enable_testing()
add_test(NAME cudf_owner_probe COMMAND cudf_owner_probe)
