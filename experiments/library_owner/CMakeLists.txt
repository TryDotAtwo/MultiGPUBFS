cmake_minimum_required(VERSION 3.30.4)
project(mgbfs_library_owner_probe LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
find_package(CUDAToolkit REQUIRED)
find_package(cudf 26.04.0 EXACT CONFIG REQUIRED)
find_package(rmm 26.04.0 EXACT CONFIG REQUIRED)
add_library(mgbfs_library_owner SHARED owner_abi.cpp layout.cu)
target_link_libraries(mgbfs_library_owner PUBLIC cudf::cudf rmm::rmm CUDA::cudart)
add_executable(cudf_owner_probe cudf_owner_probe.cpp ../../cuda/state_commit.cu)
target_link_libraries(cudf_owner_probe PRIVATE mgbfs_library_owner)
enable_testing()
add_test(NAME cudf_owner_probe COMMAND cudf_owner_probe)
add_executable(owner_abi_invalid ../../tests/library_owner_abi_invalid.cpp)
target_include_directories(owner_abi_invalid PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(owner_abi_invalid PRIVATE mgbfs_library_owner)
add_test(NAME owner_abi_invalid COMMAND owner_abi_invalid)

# Optional pinned cuco source supplied by the isolated GPU gate.
if(DEFINED CUCO_ROOT)
  add_executable(cuco_pool_probe cuco_pool_probe.cu)
  target_include_directories(cuco_pool_probe PRIVATE "${CUCO_ROOT}/include")
  target_link_libraries(cuco_pool_probe PRIVATE rmm::rmm CUDA::cudart)
  add_test(NAME cuco_pool_probe COMMAND cuco_pool_probe)
endif()
