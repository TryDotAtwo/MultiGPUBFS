===== peer_check.cu =====
#include <cuda_runtime.h>
#include <cstdio>
__global__ void smoke() {}
int main() {
  int n=0; cudaError_t e=cudaGetDeviceCount(&n);
  std::printf("device_count=%d result=%s\n",n,cudaGetErrorString(e));
  if(e!=cudaSuccess||n!=8)return 3;
  int bad=0;
  for(int a=0;a<n;++a) {
    cudaDeviceProp p{}; int coop=0;
    e=cudaGetDeviceProperties(&p,a);
    cudaError_t ce=cudaDeviceGetAttribute(&coop,cudaDevAttrCooperativeLaunch,a);
    cudaError_t se=cudaSetDevice(a);
    if(se==cudaSuccess) { smoke<<<1,1>>>();se=cudaGetLastError();if(se==cudaSuccess)se=cudaDeviceSynchronize(); }
    std::printf("gpu=%d sm=%d.%d coop=%d props=%s attr=%s smoke=%s\n",a,p.major,p.minor,coop,cudaGetErrorString(e),cudaGetErrorString(ce),cudaGetErrorString(se));
    if(e!=cudaSuccess||ce!=cudaSuccess||se!=cudaSuccess||!coop)++bad;
  }
  for(int a=0;a<n;++a) for(int b=0;b<n;++b)if(a!=b) {
    int peer=0,atomic=0;
    cudaError_t pe=cudaDeviceCanAccessPeer(&peer,a,b);
    cudaError_t ae=cudaDeviceGetP2PAttribute(&atomic,cudaDevP2PAttrNativeAtomicSupported,a,b);
    std::printf("pair=%d->%d peer=%d peer_status=%s native_atomic=%d atomic_status=%s\n",a,b,peer,cudaGetErrorString(pe),atomic,cudaGetErrorString(ae));
    if(pe!=cudaSuccess||ae!=cudaSuccess||!peer||!atomic)++bad;
  }
  std::printf("unsupported_or_failed=%d\n",bad);
  return bad?2:0;
}
